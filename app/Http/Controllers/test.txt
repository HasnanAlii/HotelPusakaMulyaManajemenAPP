<?php

namespace App\Http\Controllers;

use App\Models\Room;
use App\Models\FuzzySetting;
use Illuminate\Http\Request;

class FuzzyController extends Controller
{
    public function process(Request $request)
    {
        $request->validate([
            'harga_input'     => 'required|integer|min:50000',
            'pref_fasilitas'  => 'required|in:sedikit,cukup,lengkap',
            'pref_kenyamanan' => 'required|in:rendah,sedang,tinggi',
        ]);

        $setting = FuzzySetting::firstOrFail();

        $budget   = (int) $request->harga_input;
        $prefF    = $request->pref_fasilitas;
        $prefN    = $request->pref_kenyamanan;

        $rooms = Room::where('status', 'tersedia')->get();
        $hasil = [];

        foreach ($rooms as $room) {

            /* ==========================
               1. FUZZIFIKASI
            ========================== */
            $μHarga = $this->μHarga($room->price, $budget, $setting);
            $μFasil = $this->μFasilitas($room->facilities, $prefF, $setting);
            $μNyaman= $this->μKenyamanan(
                $this->mapKenyamanan($room->category),
                $prefN,
                $setting
            );

            /* ==========================
               2. α-PREDIKAT
            ========================== */
            $α = min($μHarga, $μFasil, $μNyaman);

            if ($α > 0) {
                $z = $setting->z_min + ($α * ($setting->z_max - $setting->z_min));

                $hasil[] = [
                    'room'  => $room,
                    'alpha' => $α,
                    'z'     => $z
                ];
            }
        }

        /* =================================================
           JIKA ADA RULE AKTIF → TSUKAMOTO NORMAL
        ================================================= */
        if (!empty($hasil)) {

            $sumAZ = 0;
            $sumA  = 0;

            foreach ($hasil as $h) {
                $sumAZ += $h['alpha'] * $h['z'];
                $sumA  += $h['alpha'];
            }

            $Zfinal = $sumAZ / $sumA;

            $rekomendasi = collect($hasil)
                ->sortByDesc('alpha')
                ->first();

            return view('fuzzy.hasil', compact(
                'hasil',
                'rekomendasi',
                'Zfinal'
            ));
        }

        /* =================================================
           FALLBACK (TIDAK ADA α > 0)
           PILIH KAMAR TERDEKAT DENGAN BUDGET
        ================================================= */
/* =================================================
   FALLBACK (TIDAK ADA α > 0)
   PILIH KAMAR TERDEKAT DENGAN BUDGET
================================================= */
$fallbackRoom = Room::where('status', 'tersedia')
    ->orderByRaw('ABS(CAST(price AS SIGNED) - ?)', [$budget])
    ->first();

return view('fuzzy.hasil', [
    'hasil' => [],
    'rekomendasi' => [
        'room'  => $fallbackRoom,
        'alpha' => 0,
        'z'     => $setting->z_min
    ],
    'Zfinal' => $setting->z_min
]);

    }

    /* ===============================
       FUZZY HARGA
    =============================== */
    private function μHarga($harga, $budget, $s)
    {
        $min = $budget * $s->harga_min_ratio;
        $mid = $budget;
        $max = $budget * $s->harga_max_ratio;

        if ($harga <= $min) return 1;
        if ($harga <= $mid) return ($harga - $min) / ($mid - $min);
        if ($harga <= $max) return ($max - $harga) / ($max - $mid);
        return 0;
    }

    /* ===============================
       FUZZY FASILITAS
    =============================== */
    private function μFasilitas($f, $pref, $s)
    {
        return match ($pref) {
            'sedikit' => $this->μLeft($f, $s->fasilitas_min, $s->fasilitas_mid),
            'cukup'   => $this->μTriangle($f, $s->fasilitas_min, $s->fasilitas_mid, $s->fasilitas_max),
            'lengkap' => $this->μRight($f, $s->fasilitas_mid, $s->fasilitas_max),
        };
    }

    /* ===============================
       FUZZY KENYAMANAN
    =============================== */
    private function μKenyamanan($n, $pref, $s)
    {
        return match ($pref) {
            'rendah' => $this->μLeft($n, $s->nyaman_min, $s->nyaman_mid),
            'sedang' => $this->μTriangle($n, $s->nyaman_min, $s->nyaman_mid, $s->nyaman_max),
            'tinggi' => $this->μRight($n, $s->nyaman_mid, $s->nyaman_max),
        };
    }

    /* ===============================
       FUNGSI KEANGGOTAAN
    =============================== */
    private function μLeft($x, $a, $b)
    {
        if ($x <= $a) return 1;
        if ($x >= $b) return 0;
        return ($b - $x) / ($b - $a);
    }

    private function μRight($x, $a, $b)
    {
        if ($x <= $a) return 0;
        if ($x >= $b) return 1;
        return ($x - $a) / ($b - $a);
    }

    private function μTriangle($x, $a, $b, $c)
    {
        if ($x <= $a || $x >= $c) return 0;
        if ($x == $b) return 1;
        if ($x < $b) return ($x - $a) / ($b - $a);
        return ($c - $x) / ($c - $b);
    }

    /* ===============================
       MAP KENYAMANAN
    =============================== */
    private function mapKenyamanan($category)
    {
        return match (strtolower($category)) {
            'standar', 'standar 1' => 1,
            'superior 1'           => 2,
            'superior 2', 'superior 3' => 3,
            default => 1,
        };
    }
}


<?php

namespace App\Http\Controllers;

use App\Models\Room;
use Illuminate\Http\Request;

class FuzzyController extends Controller
{
    public function process(Request $request)
    {
        $request->validate([
            'harga_input'     => 'required|integer|min:50000',
            'pref_fasilitas'  => 'required|in:sedikit,cukup,lengkap',
            'pref_kenyamanan' => 'required|in:rendah,sedang,tinggi',
        ]);

        $maxHarga   = (int) $request->harga_input;
        $prefFasil  = $request->pref_fasilitas;
        $prefNyaman = $request->pref_kenyamanan;

        // Sama persis dengan JS
        $rooms = Room::where('status', 'tersedia')
            ->where('price', '<=', $maxHarga)
            ->get();

        if ($rooms->isEmpty()) {
            return view('fuzzy.hasil', [
                'hasil' => [],
                'rekomendasi' => null
            ]);
        }

        $hasil = [];

        foreach ($rooms as $room) {

            /* ===============================
               FUZZY HARGA (1 / (h / 100000))
            =============================== */
            $h = $this->safeDiv(1, $this->safeDiv($room->price, 100000));

            /* ===============================
               FUZZY FASILITAS
            =============================== */
            $fasilitas = max((int)$room->facilities, 0.000001);

            if ($prefFasil === 'sedikit') {
                $f = $this->safeDiv(1, $fasilitas);
            } elseif ($prefFasil === 'cukup') {
                $f = $this->safeDiv(1, abs($fasilitas - 3) + 1);
            } else { // lengkap
                $f = $this->safeDiv($fasilitas, 5);
            }

            /* ===============================
               FUZZY KENYAMANAN
            =============================== */
            $nyaman = max($this->mapNyamanRoom($room->category), 0.000001);

            if ($prefNyaman === 'rendah') {
                $n = $this->safeDiv(1, $nyaman);
            } elseif ($prefNyaman === 'sedang') {
                $n = $this->safeDiv(1, abs($nyaman - 2) + 1);
            } else { // tinggi
                $n = $this->safeDiv($nyaman, 3);
            }

            /* ===============================
               SKOR AKHIR (JS STYLE)
            =============================== */
            $skor = ($h + $f + $n) / 3;

            $hasil[] = [
                'room' => $room,
                'skor' => $skor
            ];
        }

        usort($hasil, fn($a, $b) => $b['skor'] <=> $a['skor']);
        $rekomendasi = $hasil[0];

        return view('fuzzy.hasil', compact('hasil', 'rekomendasi'));
    }

    /* ===============================
       SAFE DIV (ANTI 0)
    =============================== */
    private function safeDiv($a, $b)
    {
        return $a / ($b == 0 ? 0.000001 : $b);
    }

    /* ===============================
       MAP KENYAMANAN (SAMA JS)
    =============================== */
    private function mapNyamanRoom($category)
    {
        return match (strtolower($category)) {
            'standar'     => 1,
            'standar 1'   => 1.5,
            'superior 1'  => 2,
            'superior 2'  => 3,
            'superior 3'  => 3,
            default       => 1,
        };
    }
}
